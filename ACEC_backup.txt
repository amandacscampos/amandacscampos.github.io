<script>
// A ACEC começa em fevereiro de 2027; antes disso, exibe "A iniciar".
function atualizarPeriodoACEC() {
    const elemento = document.getElementById('acec-periodo');
    if (!elemento) return;
    const hoje = new Date();
    const inicio = new Date(2027, 1, 1);
    const meses = (hoje.getFullYear() - inicio.getFullYear()) * 12 + hoje.getMonth() - inicio.getMonth();
    elemento.textContent = meses < 0
        ? 'fev de 2027 · A iniciar'
        : `fev de 2027 - o momento · ${meses} ${meses === 1 ? 'mês' : 'meses'}`;
}
document.addEventListener('DOMContentLoaded', atualizarPeriodoACEC);

</script>
